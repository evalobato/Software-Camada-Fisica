# 1. Dependências do sistema (PortAudio, usado pelo sounddevice)
sudo apt update
sudo apt install python3 python3-venv python3-pip libportaudio2

# 2. Ambiente virtual e bibliotecas (na raiz do repositório)
cd SOFTWARE-CAMADA-FISICA
python3 -m venv .venv
source .venv/bin/activate
pip install sounddevice numpy pytest

# 3. Executar o receptor
python3 src/receptor.py

# 4. Executar os testes automáticos
pytest tests/
